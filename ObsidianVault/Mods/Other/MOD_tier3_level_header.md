/* mod/caves/tier3_level.h -- shared layout of the UNIFIED Tier-3 level blob (Phase 3a).
 * One blob in RAM is the single source of truth for BOTH the collision cave and the render cave.
 * Generated by mod/gen_tier3_level.py (build/tier3_level.bin); written to TIER3_LEVEL_BASE live by
 * the installer (and, in Phase 3b, by a disc loader). Little-endian int32 throughout.
 *
 * header (8 words @ +0x00): magic 'T3LV', nverts, ntris, spawn x/y/z, death_y, flags
 * verts  (nverts x 3 @ +0x20):            x, y, z
 * tris   (ntris  x 7 @ +0x20 + nverts*12): i0, i1, i2, nx, ny, nz, color
 */
#ifndef TIER3_LEVEL_H
#define TIER3_LEVEL_H

/* Resident, baked-into-the-EXE address (Phase 3b): 0x8007F174 is a 514-byte ZERO-RUN *inside* the
 * loaded EXE image (offset 0x6F174 < t_size 0x72800) that is verified all-zero + never written by the
 * game (so it's loaded at boot, NOT BSS-cleared, and survives gameplay -- unlike 0x80082800 which is
 * BSS the game zeroes). The disc build (tools/build_tier3_disc.py) .incbin's the level blob here (no
 * EXE size change). The live installer writes the same blob to this address too. Blob must fit 514 B
 * (gen_tier3_level.py uses a small GRID). */
/* Default = the in-image baked location (collision-only disc, build_tier3_disc.py). The render disc
 * (build_tier3_render_disc.py) overrides this with -DTIER3_LEVEL_BASE=<DAT-append blob address>, so the
 * blob rides in the LEVEL.DAT append (read in-place) and its in-image region frees up for the loader. */
#ifndef TIER3_LEVEL_BASE
#define TIER3_LEVEL_BASE 0x8007F174
#endif
#define TIER3_LEVEL_MAGIC 0x564C3354          /* 'T3LV' */

#define LV         ((volatile int *)TIER3_LEVEL_BASE)
#define LV_MAGIC   (LV[0])
#define LV_NVERTS  (LV[1])
#define LV_NTRIS   (LV[2])
#define LV_SPAWNX  (LV[3])
#define LV_SPAWNY  (LV[4])
#define LV_SPAWNZ  (LV[5])
#define LV_DEATHY  (LV[6])

#define LV_VERTS ((volatile int *)(TIER3_LEVEL_BASE + 0x20))
#define LV_TRIS  ((volatile int *)(TIER3_LEVEL_BASE + 0x20 + LV_NVERTS * 12))

/* WALLS (blocking). Count is packed in the header flags word's high 16 bits (LV[7]); records follow the
 * triangles. Each wall = 8 int32: x0,z0,x1,z1 (XZ SEGMENT endpoints), ytop,ybot (-Y up, so ytop < ybot),
 * nx,nz (Q12 unit normal = perpendicular to the segment, *4096). FINITE: the wall query
 * (mod/caves/tier3_walls.c) blocks Flik only when ytop <= his Y <= ybot, so he can JUMP OVER it (Y above
 * ytop = clear). It does swept segment-crossing and resolves by projecting Flik (and his velocity) along
 * the stored normal -- exact for ANY wall angle. Full-height band = ytop/ybot at +-2e9.
 * ONE-MESH: these segments are AUTO-DERIVED at build time from the STEEP (wall) triangles of the level mesh
 * (mod/tier3_level_format.py: each steep face's footprint line = its two farthest XZ verts, + a unit
 * normal) -- the same steep tris are also in LV_TRIS so the render draws them as the visible 3D wall. (Plus
 * any explicit invisible blocker boxes -> 4 edge segments.) The floor query skips steep tris; only
 * walkable tris are floors. Record stride = 8 ints; keep WALL_NQ=4096 in sync with the format authority. */
#define LV_FLAGS   (LV[7] & 0xFFFF)
#define LV_NWALLS  (((unsigned)LV[7] >> 16) & 0xFFFF)
#define LV_WALLS ((volatile int *)(TIER3_LEVEL_BASE + 0x20 + LV_NVERTS * 12 + LV_NTRIS * 28))


/* vertex i -> &LV_VERTS[i*3] = {x,y,z};  triangle i -> &LV_TRIS[i*7] = {i0,i1,i2,nx,ny,nz,color} */
#define LV_VX(i) (LV_VERTS[(i) * 3 + 0])
#define LV_VY(i) (LV_VERTS[(i) * 3 + 1])
#define LV_VZ(i) (LV_VERTS[(i) * 3 + 2])

/* GAMEPLAY TRAILER (Track A) -- appended AFTER the walls so the header/vert/tri/wall offsets never shift:
 *   word0 = npickups
 *   then npickups x 4 int32: pickup (x, y, z, radius)   -- a COLLECTIBLE (drives the grain counter)
 *   then 1     x 4 int32: exit   (x, y, z, radius)       -- the GOAL volume (radius 0 = no exit) */
#define LV_TRAILER ((volatile int *)(TIER3_LEVEL_BASE + 0x20 + LV_NVERTS * 12 + LV_NTRIS * 28 + LV_NWALLS * 32))
#define LV_NPICK   (LV_TRAILER[0])
#define LV_PICKS   (LV_TRAILER + 1)                  /* pickup i -> &LV_PICKS[i*4] = {x,y,z,r} */
#define LV_EXIT    (LV_TRAILER + 1 + LV_NPICK * 4)   /* {x,y,z,r}; r<=0 -> no exit */
#define LV_PX(i)   (LV_PICKS[(i) * 4 + 0])
#define LV_PY(i)   (LV_PICKS[(i) * 4 + 1])
#define LV_PZ(i)   (LV_PICKS[(i) * 4 + 2])
#define LV_PR(i)   (LV_PICKS[(i) * 4 + 3])

/* Track C TEXTURE PALETTE (after the exit): ntextures, then each entry = texpage_raw | (clut_raw << 16).
 * A triangle's colour word top byte = its texture index; the render uses that entry's page/clut (else the
 * hardcoded dirt default). */
#define LV_NTEX    (*(LV_EXIT + 4))
#define LV_TEX     (LV_EXIT + 5)                      /* entry i = texpage | (clut << 16) */

/* Track C per-tri-vertex UVs (after the texture table): LV_NUV = 0 (render PLANAR) or ntris; then ntris*3
 * packed UVs, entry (tri i, corner k) = LV_UV[i*3 + k] = u | (v << 8). */
#define LV_NUV     (*(LV_TEX + LV_NTEX))
#define LV_UV      (LV_TEX + LV_NTEX + 1)

/* BILLBOARD props (2026-07-04, after the UV block): LV_NBPROP, then each prop = 4 int32:
 * x, y, z, (kind | size<<8). Rendered by the render cave as camera-facing textured quads. */
#define LV_NBPROP  (*(LV_UV + LV_NUV * 3))
#define LV_BPROPS  (LV_UV + LV_NUV * 3 + 1)

/* Track E spawn heading -- packed in the FLAGS word's low 16 bits (LV[7]): bit15 = "has heading", bits0-11 =
 * ABL facing angle 0..4095 (4096 = 360deg). Returns the angle to apply on spawn, or -1 = keep native default.
 * (A FIXED header read -- the trailer-end alternative made the cave's offset math overrun its code slot.) */
#define LV_SPAWN_HEADING (((LV[7] & 0x8000) ? (LV[7] & 0xFFF) : -1))

/* Shared gameplay scratch (render-region, game-free; cleared by the collision cave's spawn one-shot).
 * PICKUP_MASK = 1 bit per collected pickup (<=32); EXIT_DONE = exit triggered once. */
#define PICKUP_MASK (*(volatile int *)0x801FE010)   /* moved 0x801FB0xx->E0xx for the Track-D NODEBUF growth */
#define EXIT_DONE   (*(volatile int *)0x801FE014)   /* (keep in sync with tier3_walls.c + tier3_collision.c) */
#define HEAD_DONE   (*(volatile int *)0x801FE018)  /* Track E spawn-heading one-shot (re-armed in the spawn cave) */

/* OBJECT-OCCLUSION cache (Phase 4m): the buried-object floor query is O(ntris) and was run for EVERY on-screen
 * object EVERY frame -> the grazing-rotation lag (30->14fps). Buried status is camera-INDEPENDENT + static, so
 * amortize: floor-check ONE object per frame (round-robin OCC_CURSOR) and cache its buried bit in OCC_MASK
 * (96 bits for <=80 type-0x3f entries); each frame only PROJECT the cached-buried objects (cheap). Cleared by
 * the collision cave's spawn one-shot so a fresh level starts with no stale buried bits. */
#define OCC_CURSOR  (*(volatile int *)0x801FE01C)
#define OCC_MASK    ((volatile int *)0x801FE020)     /* OCC_MASK[0..2] = 3 x 32 bits */

#endif /* TIER3_LEVEL_H */
